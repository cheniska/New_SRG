  1. КЛАССЫ ОРУЖИЯ (по физическому носителю)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Класс — верхний уровень классификации. Определяет физическую природу снаряда,
совместимые паттерны попадания и базовое взаимодействие с защитой.

  KINETIC (Кинетика)
    Твёрдые снаряды, разгоняемые давлением или рейлганом.

  EXPLOSIVE (Взрывной)
    Снаряды с зарядом внутри, ударная волна и осколки.

  ENERGY (Энергетика)
    Направленная энергия, лазер, плазма, цепной разряд.

  1. Паттерны выстрелов
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
	точечный (за 1 ход спрайт снаряда пролетает от корабля до цели)
	пробивающий (за 1 ход спрайт снаряда пролетает пространство до вылета за пределы экрана и системы, повреждая все объекты по пути)
	дробовик (снаряды летят расходясь от корабля в форме конуса, взрываясь когда коснулись цели)
	рикошет (снаряд рикошетит до 5 раз (по одному разу за степ, случайным образом выбирая следующаю цель))
	цепной (луч/полет снаряда, отражающийся от одной цели к другой ,на каждую цель пара-тройка степов отрисовки/отрисовка полёта и отражения в 1 степ, но эффект луча дольше (до первой цели 6 степов, от первой до вторых - 5 степов, от вторых до третьих - 4, от тртетьих до четвертых 3 и т.д.))
	АоЕ (текстура рисующаяся вокруг корабля)
	точечный+Аое (первые несколько степов рисуется луч/полет снаряда, затем отрисовка аое-эффекта на месте попадания)
	Затухающий - большая дальность, спрайт летит от корабля до цели, чем больше расстояние тем меньше урон
	луч - отрисовывается отрезком с текстурой между кораблём и целью
	самонаводящийся (ракеты и торпеды, имеют своё небольшое хп и скорость, торпеды мощне и если цель убита раньше достидения её торпедой, торпеда возвращается к носителю)
	мины (отличаются от ракет неподвижностью, могут быть с таймером, подрывом по команде, магнитящиеся в небольшом радиусе)

  2. ТИПЫ УРОНА
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Физическая природа урона. Определяет взаимодействие с бронёй, щитом и
восприимчивостью корпуса. Тип урона не равен классу оружия — ракета может
нести взрывной, ЭМИ или кинетический заряд.

  ┌─────────────────┬
  │ Тип урона       │
  ├─────────────────┼
  │ Кинетический    │
  │ Взрывной        │
  │ Энергетический  │
  └─────────────────┴

Примечание: «Прямой урон» (игнорирует броню и щит) — не отдельный тип, а
комбинация свойств снаряда ArmorPenetration=MAX + ShieldPenetration=MAX.

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
  4. СВОЙСТВА СНАРЯДА
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Параметры конкретного оружия, не зависящие от типа урона.
Задаются числами — дают градацию от частичного до полного эффекта.

  ArmorPenetration    Снижает или полностью игнорирует плоское поглощение брони.
                      Значение: 0..MAX (единицы брони, которые игнорируются).

  ShieldPenetration   Процент урона, проходящего сквозь щит без трат прочности.
                      Значение: 0..100%.

  EquipDamage         Наносит урон случайному или конкретному модулю оборудования.
                      Используется совместно с основным уроном или отдельно.

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
  5. ЗАЩИТА: БРОНЯ И ЩИТ
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  БРОНЯ
    Плоское поглощение в единицах (напр. 50 ед.) — срезает фиксированную часть
    каждого удара. Мелкое оружие становится бесполезным против тяжёлых кораблей.
    Мягкий кап: броня поглощает X ед., но не более Y% от удара.

  ЩИТ
    Блокирует фиксированный % урона, пока не сломается. Не вторая полоска HP —
    именно процентный барьер с параметром прочности (Capacity) и регенерацией
    (RegenRate). Энергетический урон перегружает быстрее.


━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
  6. ВОСПРИИМЧИВОСТЬ КОРПУСА
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Каждый тип корпуса имеет процентные модификаторы к типам урона.
Применяются до вычитания брони.

У каждого корпуса количество восприимчивостей - полное, то есть есть своя восприимчивость к каждому типу урона

  7. ФОРМУЛА РАСЧЁТА УРОНА
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  Итоговый урон =
      BaseDamage
    × HullSusceptibility[DamageType]         // восприимчивость корпуса
    − max(Armor − ArmorPenetration, 0)        // броня за вычетом пробития
      но не более ArmorCap% от удара
    × (1 − ShieldBlock%)                     // если щит цел
      × (1 − ShieldPenetration)              // и снаряд не проходит насквозь

  8. ЭФФЕКТЫ И СТАТУСЫ
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Эффекты — дополнительные свойства конкретных орудий.
Могут также навешиваться артефактами или модификациями оборудования.
Применяются поверх урона и комбинируются между собой.

  БОЕВЫЕ ЭФФЕКТЫ (непосредственное влияние на бой)
  ─────────────────────────────────────────────────
  Slow            Снижает скорость и манёвренность цели.
                  Градации: лёгкое → сильное → обездвиживание.
                  Стекается до капа.

  Shutdown        Временно отключает все системы корабля на N ходов с некоторым шансом сломать 		   оборудование. (ЭМИ)

  ArmorDebuff     Снижает броню цели на N единиц временно.
                  Стекается до капа. Тактика: сначала дебафф — потом кинетика. (корозия)

  Jamming         ЭМ-помехи: снижает дальность радара, оружия и мощность
                  сканера цели.

  EngineDisable   Прямой урон двигателю как модулю. Снижает скорость
                  пропорционально повреждению. Выделен отдельно от EquipDamage
                  из-за немедленного тактического эффекта на позиционирование.

  Drain           Восстанавливает HP атакующего за нанесённый урон (вампиризм).

  ТАКТИЧЕСКИЕ ЭФФЕКТЫ (требуют условия: сканер > защита цели)
  ─────────────────────────────────────────────────────────────
  Следующие эффекты активируются только если мощность сканера атакующего
  превышает защиту цели, и цель находится в радиусе радара.

  BlockWeapon     Блокирует оружие цели на N ходов. Шанс обратно пропорционален
                  размеру корпуса. Детерминирован при выполнении условия сканера.

  BlockDroid      Блокирует дроида-ремонтника цели на N ходов. Цель перестаёт
                  регенерировать HP в течение эффекта.
                  Дроид — внешний механизм починки обшивки корабля.

  ExecuteBonus    Увеличивает урон по повреждённой цели до +33% (если у цели
                  остался 1 HP) пропорционально степени повреждения.
                  Формула: k = k × (1 + 0.33 × (1 − CurrentHP / MaxHP))

  ЭКОНОМИЧЕСКИЕ ЭФФЕКТЫ
  ──────────────────────
  LootFocus       Повышает шанс дропа оборудования при уничтожении цели.
                  Нарратив: оружие разрушает корпус аккуратно, не взрывая
                  внутренности — хирургический удар. Компромисс: низкий DPS.
                  Условие активации: цель уничтожена в этом бою.

эффект может быть наложен на выстрелы сторонним предметом, например артефактом, а может быть родным у пушки или её модификации
