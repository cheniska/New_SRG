2. ГЛОБАЛЬНЫЕ НАСТРОЙКИ
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

В GameSettingsConfig.cs добавить:

    [Header("Equipment")]
    public bool AllowEquipmentChangeInSpace = false;

Если false — UI снаряжения доступен только при стыковке с объектом у которого
есть признак HasDock (планета, станция, корабль-носитель).
Проверяется в UI-контроллере, не в симуляции.

3. ОБЩИЕ ХАРАКТЕРИСТИКИ ЛЮБОГО ПРЕДМЕТА
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Каждый предмет независимо от типа имеет:

  Name                  — отображаемое название (string)
  Weight                — вес (int), влияет на ActualSpeed
  Price                 — базовая цена покупки (int)
  Durability            — текущая прочность предмета (int)
  MaxDurability         — максимальная прочность (int); при 0 предмет сломан
                          и не работает до ремонта
  Manufacturer.Race     — раса-производитель (string)
  Manufacturer.Faction  — фракция-производитель (string)
  TechLevel             — технический уровень (int, 1..N); предмет не появляется
                          в магазинах/луте если TechLevel > TechLevel локации
  Modification          — ID текущей модификации (string, nullable)
  Modules               — список установленных дополнительных модулей ([]string)

4. ТЕХНИЧЕСКИЙ УРОВЕНЬ
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

4.1 Конфиг уровней
В GalaxyConfig.json добавить:

    "TechLevels": {
        "Count": 5,
        "Names": ["Примитивный", "Базовый", "Развитый", "Передовой", "Легендарный"]
    }

4.2 Галактический уровень
В GalaxyData добавить поле GalaxyTechLevel (int). Задаётся при генерации.
Предметы с TechLevel > GalaxyTechLevel не генерируются нигде в галактике.

4.3 Планетарный уровень
В PlanetData и SatelliteData добавить поле TechLevel (int).
Определяет максимальный уровень товаров в магазине и качество лута.
Планета не может торговать предметами выше своего TechLevel.

5. СЕРИИ КОРПУСОВ И СИСТЕМА СЛОТОВ
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

5.1 Принцип
Корпус принадлежит серии. Серия определяет:
- Набор и количество слотов (не превышая глобальные лимиты GalaxyConfig.Equipment)
- Восприимчивость к типам урона (WeaponVulnerability)

Базовые слоты (всегда 1, не зависят от серии):
  Engine_0, FuelTank_0

Серия управляет только вариативными слотами. Корпус без серии имеет все слоты

5.4 Восприимчивость к урону
WeaponVulnerability задаётся на серии, не на конкретном корпусе.
Дефолт — 1.0 по всем типам если серия не задаёт значение.
Типы урона: Kinetic, Energy, Missile. Расширяется в конфиге.

6. ПАРАМЕТРЫ ПО ТИПАМ ОБОРУДОВАНИЯ
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

6.1 Engine — Двигатель
  Speed       — базовая скорость внутри системы (int)
  JumpRange   — дальность прыжка между системами (int); 0 = прыжки недоступны

6.2 FuelTank — Топливный бак
  Capacity    — максимальное количество топлива (int)
  CurrentFuel — текущее топливо (int, рантайм); в конфиге = Capacity
                тратится на прыжки и форсаж

6.3 Shield — Генератор защиты
  (переименованы по сравнению с текущим конфигом)
  BlockPercent  — % входящего урона поглощаемый щитом (int, 0–100)
                  остаток идёт в броню → HP

6.4 Radar — Радар
  Range  — дальность обнаружения (int)

6.5 Scaner — Сканер
  (переименован: Depth → Power)
  Power  — мощность сканирования (int)
           сканирование успешно если Power > мощности генератора защиты цели

6.6 Droid — Ремонтный дроид
  HealPerTurn  — HP корпуса восстанавливаемых за ход (int)

6.7 CargoGrabber — Грузовой захват
  Power        — мощность захвата (int); должна превышать сопротивление объекта
  Range        — максимальная дистанция захвата (int)
  Speed — скорость притягивания объекта в единицах за Step, объект с максимального расстояния всегда успевает притянуться в течение хода1
                 объект движется к кораблю порциями, не мгновенно

6.8 Weapons — Оружие
  MinDmg      — минимальный урон (int)
  MaxDmg      — максимальный урон (int)
  Range       — дальность стрельбы (int)
  DamageType  — тип урона: Kinetic / Energy / Missile (string)

6.10 Artefacts — Артефакты
  Особая категория: уникальные эффекты, параметры задаются свободно через Params.
  Примеры:
    AF_SpeedBoost — SpeedMult: 1.2 (множитель к ActualSpeed)
    AF_HullRepair — HealPerTurn: 3 (дополнительное восстановление HP)
артефакты могут быть и сложными, активируемыми, и активными (например, сбивать ракеты)

7. ФОРМУЛА РАСЧЁТА УРОНА
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Последовательность при попадании:

  1. effectiveDamage = rawDamage × WeaponVulnerability[DamageType]
  2. shieldAbsorbed  = effectiveDamage × (BlockPercent / 100)
                       вычитается из текущего заряда щита
  3. afterShield     = effectiveDamage − shieldAbsorbed
  4. afterArmor      = max(0, afterShield − (Armor − ArmorPierce))
  5. afterArmor применяется к CurrentHull

Реализация: метод ApplyDamage(int rawDamage, string damageType, int armorPierce)
в ShipData.

9. СИСТЕМА СЛОТОВ — ИМЕНОВАНИЕ И ЛОГИКА
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Именование слота: CategoryName_Index
Примеры: Weapon_0, Weapon_1, Shield_0, Artifact_2

Базовые слоты (всегда, независимо от серии):
  Engine_0, FuelTank_0

Вариативные слоты — берутся из HullSeries[series].Slots.
Количество не превышает GalaxyConfig.Equipment[category].Count.
В серии всегда преечисляются все вариативные слоты

Операции:
- Установка: itemId из Inventory → Equipment[slot]
  Если слот занят — предметы меняются местами.
- Снятие: itemId из Equipment[slot] → Inventory
- Смена корпуса: если новая серия имеет меньше слотов —
  лишнее оборудование выгружается в Inventory автоматически.

! Если AllowEquipmentChangeInSpace = false и корабль не пристыкован
  (HasDock у текущего объекта) — операции установки/снятия блокируются в UI.

10. СТАНЦИИ — ОБОРУДОВАНИЕ
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Станции используют тот же механизм Equipment что и корабли:
